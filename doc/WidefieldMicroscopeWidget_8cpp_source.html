351 void WidefieldMicroscopeWidget::UpdateWidefieldUIData(Util::SynchronizedPointer<WidefieldMicroscopeData>& ModuleData)
352 {
353 if (!ui->SBWidefieldLEDExposureTime->hasFocus())
354 ui->SBWidefieldLEDExposureTime->setValue(ModuleData->GetLEDCameraExposureTime().count());
355
356 if (!ui->SBWidefieldPumpExposureTime->hasFocus())
357 ui->SBWidefieldPumpExposureTime->setValue(ModuleData->GetWidefieldCameraExposureTime().count());
358
359 if (ModuleData->IsCurrentImageAvlbl())
360 {
361 // SetMainGraphicsImage creates a deep copy of the current image.
362 SetMainGraphicsImage(ModuleData->GetCurrentImage());
363 ModuleData->ResetCurrentImageAvlbl();
364 }
365
366 if (!SaveWidefieldImageFilename.isEmpty())
367 {
368 ModuleData->EnqueueEvent(DynExp::MakeEvent(&GetCastOwner(), &WidefieldMicroscope::OnSaveCurrentImage, SaveWidefieldImageFilename));
369
370 SaveWidefieldImageFilename.clear();
371 }
372
373 if (ModuleData->HaveLocalizedPositionsChanged())
374 {
375 MainGraphicsView->RemoveMarkers(true);
376 for (const auto& Position : ModuleData->GetLocalizedPositions())
377 MainGraphicsView->AddMarker(Position.second.Position, QColorConstants::DarkYellow, true, Position.first, "Emitter " + Util::ToStr(Position.first));
378
379 ModuleData->ResetLocalizedPositionsChanged();
380 }
381
382 if (CurrentConfocalSpotPosition != ModuleData->GetConfocalSpotImagePosition())
383 {
384 MainGraphicsView->RemoveMarker(CurrentConfocalSpotPosition);
385 MainGraphicsView->AddMarker(ModuleData->GetConfocalSpotImagePosition(), QColorConstants::Red, false, -1, "Confocal spot");
386
387 CurrentConfocalSpotPosition = ModuleData->GetConfocalSpotImagePosition();
388 }
389
390 if (MainGraphicsView->HaveMarkersChanged())
391 {
392 // Copy intended.
393 const auto LocalizedPositions = ModuleData->GetLocalizedPositions();
394 ModuleData->ClearLocalizedPositions();
395
396 bool SortingEnabled = ui->TWEmitterList->isSortingEnabled();
397 ui->TWEmitterList->setSortingEnabled(false);
398
399 ui->TWEmitterList->clearContents();
400 ui->TWEmitterList->setRowCount(0);
401
402 auto& Markers = MainGraphicsView->GetMarkers();
403 for (const auto& Marker : Markers)
404 {
405 const auto Row = ui->TWEmitterList->rowCount();
406 ui->TWEmitterList->insertRow(Row);
407
408 QTableWidgetItem* IDItem = new QTableWidgetItem;
409 WidefieldMicroscopeData::LocalizedEmitterStateType LocalizedEmitterState = WidefieldMicroscopeData::LocalizedEmitterStateType::NotSet;
410 if (Marker.GetID() >= 0)
411 {
412 IDItem->setData(Qt::EditRole, Marker.GetID());
413 IDItem->setData(Qt::UserRole, Marker.GetID());
414
415 auto OldLocalizedPosition = LocalizedPositions.find(Marker.GetID());
416 if (OldLocalizedPosition != LocalizedPositions.cend())
417 LocalizedEmitterState = OldLocalizedPosition->second.State;
418 ModuleData->AppendLocalizedPosition({ Marker.GetID(), { Marker.GetMarkerPos(), LocalizedEmitterState } });
419 }
420 else
421 {
422 IDItem->setData(Qt::UserRole, Util::MarkerGraphicsView::MarkerType::IDType(-1));
423 IDItem->setText("n/a");
424 }
425
426 ui->TWEmitterList->setItem(Row, EmitterListColumnType::EmitterID, IDItem);
427 ui->TWEmitterList->setItem(Row, EmitterListColumnType::EmitterName, new QTableWidgetItem(Marker.GetName().data()));
428 ui->TWEmitterList->setItem(Row, EmitterListColumnType::Image_x, new QTableWidgetItem(QString::number(Marker.GetMarkerPos().x())));
429 ui->TWEmitterList->item(Row, EmitterListColumnType::Image_x)->setData(Qt::ItemDataRole::UserRole, Marker.GetMarkerPos().x());
430 ui->TWEmitterList->setItem(Row, EmitterListColumnType::Image_y, new QTableWidgetItem(QString::number(Marker.GetMarkerPos().y())));
