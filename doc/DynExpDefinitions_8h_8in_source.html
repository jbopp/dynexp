189constexpr int CompilationYear()
190{
191 constexpr int YearChar = 7;
192 constexpr int NumYearChars = 4;
193
194 int result = 0;
195 for (int i = YearChar + NumYearChars - 1, Multiplier = 1; i >= YearChar; i--, Multiplier *= 10)
196 result += (__DATE__[i] - '0') * Multiplier;
197
198 return result;
199}
