207 void ODMRWidget::OnSweepSeriesParamChanged(int Index)
208 {
209 QDoubleSpinBox* Destiny = nullptr;
210
211 if (Index != 2) // != ODMRData::SweepSeriesType::AnalogOut
212 {
213 switch (Index)
214 {
215 case 0: // ODMRData::SweepSeriesType::RFModulationDepth
216 Destiny = ui->SBRFModulationDepth;
217 break;
218 case 1: // ODMRData::SweepSeriesType::RFPower
219 Destiny = ui->SBRFPower;
220 break;
221 }
222
223 if (Destiny)
224 {
225 ui->SBParamSweepStart->setMinimum(Destiny->minimum());
226 ui->SBParamSweepStart->setMaximum(Destiny->maximum());
227 ui->SBParamSweepStart->setSuffix(Destiny->suffix());
228 ui->SBParamSweepStop->setMinimum(Destiny->minimum());
229 ui->SBParamSweepStop->setMaximum(Destiny->maximum());
230 ui->SBParamSweepStop->setSuffix(Destiny->suffix());
231 ui->SBParamSweepStep->setMinimum(Destiny->minimum());
232 ui->SBParamSweepStep->setMaximum(Destiny->maximum());
233 ui->SBParamSweepStep->setSuffix(Destiny->suffix());
234 }
235 }
236 else
237 {
238 ui->SBParamSweepStart->setMinimum(AuxAnalogOutMinValue);
239 ui->SBParamSweepStart->setMaximum(AuxAnalogOutMaxValue);
240 ui->SBParamSweepStart->setSuffix(QString(" ") + DynExp::Units::UnitTypeToStr(AuxAnalogOutValueUnit));
241 ui->SBParamSweepStop->setMinimum(AuxAnalogOutMinValue);
242 ui->SBParamSweepStop->setMaximum(AuxAnalogOutMaxValue);
243 ui->SBParamSweepStop->setSuffix(ui->SBParamSweepStart->suffix());
244 ui->SBParamSweepStep->setMinimum(AuxAnalogOutMinValue);
245 ui->SBParamSweepStep->setMaximum(AuxAnalogOutMaxValue);
246 ui->SBParamSweepStep->setSuffix(ui->SBParamSweepStart->suffix());
247 }
